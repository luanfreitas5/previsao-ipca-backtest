    def assert_no_leakage(context: ForecastContext) -> None:
    """Falha se o contexto contiver qualquer informação posterior à origem.

    Verifica (1) IPCA no histórico divulgado após a origem, (2) alvos de treino
    divulgados após a origem e (3) linhas de teste de outra origem.

    Parameters
    ----------
    context : ForecastContext
        Contexto a verificar.

    Raises
    ------
    LeakageError
        Se qualquer verificação falhar.

    Examples
    --------
    >>> assert_no_leakage(context)  # doctest: +SKIP
    """
    latest_history = cast("date | None", context.ipca_history["available_at"].max())
    if latest_history is not None and latest_history > context.origin:
        raise LeakageError(f"IPCA divulgado em {latest_history} usado na origem {context.origin}")
    latest_target = (
        cast("date | None", context.train["target_available_at"].max())
        if context.train.height
        else None
    )
    if latest_target is not None and latest_target > context.origin:
        raise LeakageError(f"Alvo de treino divulgado em {latest_target} > origem {context.origin}")
    if context.test.height and context.test["origin"].n_unique() != 1:
        raise LeakageError("Linhas de teste de mais de uma origem no mesmo contexto")

    def build_context(
    origin: date, ledger_ipca: pl.DataFrame, supervised: pl.DataFrame, horizons: Sequence[int]
) -> tuple[ForecastContext, pl.DataFrame]:
    """Monta o contexto de uma origem e separa o gabarito (alvos reais) do que o modelo vê.

    Parameters
    ----------
    origin : date
        Data de origem.
    ledger_ipca : pl.DataFrame
        Linhas do ledger com ``series == "ipca"``.
    supervised : pl.DataFrame
        Tabela supervisionada completa.
    horizons : Sequence[int]
        Horizontes avaliados.

    Returns
    -------
    tuple[ForecastContext, pl.DataFrame]
        Contexto (sem ``y_true`` na parte de teste) e o gabarito da origem.

    Raises
    ------
    LeakageError
        Se o contexto montado contiver informação futura.

    Examples
    --------
    >>> context, truth = build_context(
    ...     date(2020, 1, 31), ipca, supervised, [1, 3]
    ... )  # doctest: +SKIP
    """
    test_full = supervised.filter(pl.col("origin") == origin)
    context = ForecastContext(
        origin=origin,
        horizons=tuple(horizons),
        ipca_history=ledger_ipca.filter(pl.col("available_at") <= origin).sort("period"),
        train=supervised.filter(pl.col("target_available_at") <= origin),
        test=test_full.drop("y_true", "target_available_at"),
    )
    assert_no_leakage(context)
    truth = test_full.select("horizon", "target_period", "target_available_at", "y_true")
    return context, truth

    def run_walk_forward(
    supervised: pl.DataFrame,
    ledger: pl.DataFrame,
    models: Sequence[Forecaster],
    origins: Sequence[date],
    show_progress: bool = True,
) -> pl.DataFrame:
    """Executa o backtest com janela expansiva: a cada origem, reajusta e prevê.

    Parameters
    ----------
    supervised : pl.DataFrame
        Tabela supervisionada (``build_supervised_frame``).
    ledger : pl.DataFrame
        Ledger point-in-time.
    models : Sequence[Forecaster]
        Modelos a avaliar (incluindo o baseline Focus).
    origins : Sequence[date]
        Origens a avaliar, em ordem crescente.
    show_progress : bool, optional
        Exibe barra de progresso rich, by default ``True``.

    Returns
    -------
    pl.DataFrame
        Previsões em formato longo (``model``, ``origin``, ``horizon``, ``target_period``,
        ``target_available_at``, ``y_true``, ``y_pred``); ``y_pred`` é nulo quando o
        modelo falhou ou não tinha dados.

    Raises
    ------
    LeakageError
        Se algum contexto contiver informação futura.

    Examples
    --------
    >>> run_walk_forward(supervised, ledger, models, origins)  # doctest: +SKIP
    """
    horizons = sorted(supervised["horizon"].unique().to_list())
    ipca = ledger.filter(pl.col("series") == "ipca")
    rows: list[dict[str, Any]] = []
    progress = Progress(
        SpinnerColumn(),
        TextColumn("[bold blue]{task.description}"),
        BarColumn(),
        MofNCompleteColumn(),
        TaskProgressColumn(),
        TimeElapsedColumn(),
        TimeRemainingColumn(),
        console=CONSOLE,
        disable=not show_progress,
    )
    with progress:
        task = progress.add_task("Backtest walk-forward", total=len(origins))
        for origin in origins:
            context, truth = build_context(origin, ipca, supervised, horizons)
            progress.advance(task)
            if truth.drop_nulls("y_true").is_empty():
                continue
            for model in models:
                rows.extend(_score_model(model, context, truth))
    result = pl.DataFrame(rows, schema=_RESULT_SCHEMA)
    logger.info("Backtest concluído: %d previsões, %d origens", result.height, len(origins))
    return validate_dataframe(result, FORECASTS_SCHEMA, "forecasts")

    def add_relative_rmse(metrics: pl.DataFrame, reference: str) -> pl.DataFrame:
    """Adiciona ``rmse_ratio`` = RMSE do modelo / RMSE do modelo de referência (por horizonte).

    Valores acima de 1 indicam que o modelo perde para a referência.

    Parameters
    ----------
    metrics : pl.DataFrame
        Saída de ``compute_metrics``.
    reference : str
        Nome do modelo de referência (ex.: ``"focus"``).

    Returns
    -------
    pl.DataFrame
        ``metrics`` com a coluna ``rmse_ratio``.

    Examples
    --------
    >>> add_relative_rmse(metrics, "focus")  # doctest: +SKIP
    """
    ref = metrics.filter(pl.col("model") == reference).select(
        "horizon", pl.col("rmse").alias("_ref_rmse")
    )
    return (
        metrics.join(ref, on="horizon", how="left")
        .with_columns((pl.col("rmse") / pl.col("_ref_rmse")).alias("rmse_ratio"))
        .drop("_ref_rmse")
    )

    def block_bootstrap_means(
    values: np.ndarray, block_length: int, n_boot: int, rng: np.random.Generator
) -> np.ndarray:
    """Médias de reamostras por bootstrap em blocos móveis (preserva autocorrelação).

    Erros de previsão a ``h`` passos são autocorrelacionados até ``h-1`` defasagens;
    reamostrar observações isoladas subestimaria a incerteza.

    Parameters
    ----------
    values : np.ndarray
        Série ordenada no tempo.
    block_length : int
        Tamanho do bloco (use o horizonte).
    n_boot : int
        Número de reamostras.
    rng : np.random.Generator
        Gerador aleatório (semente fixa para reprodutibilidade).

    Returns
    -------
    np.ndarray
        Vetor com ``n_boot`` médias reamostradas.

    Examples
    --------
    >>> block_bootstrap_means(np.arange(10.0), 2, 100, np.random.default_rng(0)).shape
    (100,)
    """
    n = len(values)
    block = max(1, min(block_length, n))
    n_blocks = math.ceil(n / block)
    starts = rng.integers(0, n - block + 1, size=(n_boot, n_blocks))
    index = (starts[:, :, None] + np.arange(block)).reshape(n_boot, -1)[:, :n]
    return values[index].mean(axis=1)

    def compute_metrics(
    errors: pl.DataFrame, n_bootstrap: int = 1000, seed: int = 42, alpha: float = 0.05
) -> pl.DataFrame:
    """Calcula MAE, RMSE e viés por (modelo, horizonte) com IC de bootstrap em blocos.

    Parameters
    ----------
    errors : pl.DataFrame
        Saída de ``prepare_errors`` (idealmente na amostra comum).
    n_bootstrap : int, optional
        Reamostras do bootstrap, by default 1000.
    seed : int, optional
        Semente do bootstrap, by default 42.
    alpha : float, optional
        Nível do IC (0.05 = IC de 95%), by default 0.05.

    Returns
    -------
    pl.DataFrame
        ``model``, ``horizon``, ``n``, ``mae``, ``mae_lo``, ``mae_hi``, ``rmse``,
        ``rmse_lo``, ``rmse_hi``, ``bias``.

    Examples
    --------
    >>> compute_metrics(errors).columns[:4]  # doctest: +SKIP
    ['model', 'horizon', 'n', 'mae']
    """
    rng = np.random.default_rng(seed)
    rows = []
    for group in errors.sort("origin").partition_by(["model", "horizon"], maintain_order=True):
        horizon = int(group["horizon"][0])
        abs_err = group["abs_error"].to_numpy()
        sq_err = group["sq_error"].to_numpy()
        mae_lo, mae_hi = np.quantile(
            block_bootstrap_means(abs_err, horizon, n_bootstrap, rng), [alpha / 2, 1 - alpha / 2]
        )
        mse_lo, mse_hi = np.quantile(
            block_bootstrap_means(sq_err, horizon, n_bootstrap, rng), [alpha / 2, 1 - alpha / 2]
        )
        rows.append(
            {
                "model": group["model"][0],
                "horizon": horizon,
                "n": group.height,
                "mae": float(abs_err.mean()),
                "mae_lo": float(mae_lo),
                "mae_hi": float(mae_hi),
                "rmse": float(np.sqrt(sq_err.mean())),
                "rmse_lo": float(np.sqrt(mse_lo)),
                "rmse_hi": float(np.sqrt(mse_hi)),
                "bias": float(group["error"].mean()),  # type: ignore[arg-type]
            }
        )
    return pl.DataFrame(rows).sort("horizon", "model")

    def compute_metrics_by_slice(
    errors: pl.DataFrame, slices: Sequence[tuple[str, date, date]]
) -> pl.DataFrame:
    """RMSE e MAE por (fatia, modelo, horizonte), com a fatia definida pelo mês-alvo.

    Métricas agregadas escondem períodos em que o modelo falha; a fatia
    evidencia, por exemplo, o choque inflacionário de 2021-2022.

    Parameters
    ----------
    errors : pl.DataFrame
        Saída de ``prepare_errors``.
    slices : Sequence[tuple[str, date, date]]
        ``(nome, início, fim)`` de cada subperíodo (inclusive).

    Returns
    -------
    pl.DataFrame
        ``slice``, ``model``, ``horizon``, ``n``, ``mae``, ``rmse``.

    Examples
    --------
    >>> compute_metrics_by_slice(
    ...     errors, [("2023+", date(2023, 1, 1), date(2100, 1, 1))]
    ... )  # doctest: +SKIP
    """
    parts = []
    for name, start, end in slices:
        subset = errors.filter(pl.col("target_period").is_between(start, end))
        if subset.is_empty():
            continue
        parts.append(
            subset.group_by("model", "horizon")
            .agg(
                pl.len().alias("n"),
                pl.col("abs_error").mean().alias("mae"),
                pl.col("sq_error").mean().sqrt().alias("rmse"),
            )
            .with_columns(pl.lit(name).alias("slice"))
            .select("slice", "model", "horizon", "n", "mae", "rmse")
        )
    if not parts:
        return pl.DataFrame(
            schema={
                "slice": pl.String,
                "model": pl.String,
                "horizon": pl.Int64,
                "n": pl.UInt32,
                "mae": pl.Float64,
                "rmse": pl.Float64,
            }
        )
    return pl.concat(parts).sort("slice", "horizon", "model")

    def diebold_mariano(
    errors_a: np.ndarray,
    errors_b: np.ndarray,
    horizon: int = 1,
    loss: Loss = "squared",
    alternative: Alternative = "two-sided",
) -> DMResult:
    """Testa H0: os modelos A e B têm a mesma acurácia preditiva.

    O diferencial ``d_t = L(e_A,t) - L(e_B,t)`` tem autocorrelação até ``h-1`` em
    previsões a ``h`` passos; a variância usa essas autocovariâncias e a estatística
    recebe a correção de amostra pequena de Harvey et al. (1997), com cauda t(n-1).

    Parameters
    ----------
    errors_a, errors_b : np.ndarray
        Erros de previsão dos modelos A e B, pareados e ordenados no tempo.
    horizon : int, optional
        Horizonte de previsão ``h`` (define quantas autocovariâncias entram), by default 1.
    loss : {"squared", "absolute"}, optional
        Função de perda, by default ``"squared"``.
    alternative : {"two-sided", "less", "greater"}, optional
        ``"less"``: A é mais acurado que B; ``"greater"``: A é menos acurado.

    Returns
    -------
    DMResult
        Estatística, valor-p, diferença média e tamanho amostral. ``nan`` se a
        amostra for curta demais ou o diferencial for degenerado (variância <= 0).

    Raises
    ------
    ValueError
        Se os vetores tiverem tamanhos diferentes ou ``horizon < 1``.

    Examples
    --------
    >>> rng = np.random.default_rng(0)
    >>> a, b = rng.normal(0, 1, 100), rng.normal(0, 2, 100)
    >>> diebold_mariano(a, b, horizon=1, alternative="less").p_value < 0.05
    True
    """
    if len(errors_a) != len(errors_b):
        raise ValueError("Os vetores de erros devem ter o mesmo tamanho")
    if horizon < 1:
        raise ValueError("horizon deve ser >= 1")
    n = len(errors_a)
    func = np.square if loss == "squared" else np.abs
    diff = func(errors_a) - func(errors_b)
    mean_diff = float(diff.mean()) if n else math.nan
    if n <= 2 * horizon:
        return DMResult(math.nan, math.nan, mean_diff, n)
    variance = _long_run_variance(diff, horizon)
    if variance <= 0:
        return DMResult(math.nan, math.nan, mean_diff, n)
    correction = math.sqrt((n + 1 - 2 * horizon + horizon * (horizon - 1) / n) / n)
    statistic = correction * mean_diff / math.sqrt(variance)
    dist = stats.t(df=n - 1)
    p_value = {
        "two-sided": 2 * dist.sf(abs(statistic)),
        "less": dist.cdf(statistic),
        "greater": dist.sf(statistic),
    }[alternative]
    return DMResult(float(statistic), float(p_value), mean_diff, n)

        pl.col("survey_date"),
        pl.col("survey_date").dt.offset_by(f"{publication_lag_days}d").alias("focus_available_at"),

    def fetch_sgs(self, code: int, start: date, end: date) -> list[dict[str, str]]:
        """Baixa uma série SGS em janelas de ``chunk_years`` anos.

        A API limita a janela de consulta de séries diárias, por isso o período
        é fatiado em blocos consecutivos e sem sobreposição.

        Parameters
        ----------
        code : int
            Código da série no SGS.
        start : date
            Data inicial (inclusive).
        end : date

        Returns

        list[dict[str, str]]

        Raises
        ------
        IngestionError
            Se a API falhar após todas as tentativas.
        """
        url = f"{self._config.sgs_base_url}/bcdata.sgs.{code}/dados"
        records: list[dict[str, str]] = []
        for window_start, window_end in self._build_windows(start, end):
            params = {
                "formato": "json",
                "dataInicial": window_start.strftime("%d/%m/%Y"),
                "dataFinal": window_end.strftime("%d/%m/%Y"),
            }
            payload = self._get_json(url, params)
            if not isinstance(payload, list):
                raise IngestionError(f"Resposta inesperada do SGS {code}: {payload!r}")
            records.extend(payload)
        logger.info("SGS %s: %d registros baixados", code, len(records))
        return records

    def fetch_focus(self, indicator: str, base_calculo: int) -> list[dict[str, Any]]:
        """Baixa as expectativas mensais do Focus (Olinda) com paginação OData.

        Parameters
        ----------
        indicator : str
            Indicador (ex.: ``"IPCA"``).
        base_calculo : int
            0 = estatísticas dos últimos 30 dias; 1 = últimos 5 dias úteis.

        Returns
        -------
        list[dict[str, Any]]
            Registros exatamente como recebidos.

        Raises
        ------
        IngestionError
            Se a API falhar após todas as tentativas.
        """
        url = f"{self._config.olinda_url}/ExpectativaMercadoMensais"
        page_size = self._config.focus_page_size
        records: list[dict[str, Any]] = []
        skip = 0
        while True:
            params = {
                "$filter": f"Indicador eq '{indicator}' and baseCalculo eq {base_calculo}",
                "$select": FOCUS_FIELDS,
                "$orderby": "Data asc,DataReferencia asc",
                "$format": "json",
                "$top": str(page_size),
                "$skip": str(skip),
            }
            payload = self._get_json(url, params)
            page = payload.get("value", []) if isinstance(payload, dict) else None
            if page is None:

    def _build_windows(self, start: date, end: date) -> list[tuple[date, date]]:
        """Fatia ``[start, end]`` em janelas consecutivas de ``chunk_years`` anos."""
        windows: list[tuple[date, date]] = []
        cursor = start
        while cursor <= end:
            next_start = add_months(cursor.replace(day=1), 12 * self._config.chunk_years)
            window_end = min(next_start - timedelta(days=1), end)
            windows.append((cursor, window_end))
            cursor = window_end + timedelta(days=1)
        return windows

    def _fetch_json_once(self, url: str, params: dict[str, str]) -> Any:
        """Executa um único GET; levanta ``HTTPStatusError`` em status de erro/retentável."""
        # O Olinda não aceita "+" como espaço (httpx o usa por padrão): exige %20
        response = self._client.get(f"{url}?{urlencode(params, quote_via=quote)}")
        if response.status_code in _RETRYABLE_STATUS:
            raise httpx.HTTPStatusError(
                f"status {response.status_code}", request=response.request, response=response
            )
        response.raise_for_status()
        return response.json()

    def _get_json(self, url: str, params: dict[str, str]) -> Any:
        """GET com retry exponencial em erros transitórios (rede, 429, 5xx)."""
        attempts = self._config.max_retries + 1
        last_error: Exception | None = None
        for attempt in range(attempts):
            try:
                return self._fetch_json_once(url, params)
            except (httpx.TransportError, httpx.HTTPStatusError, ValueError) as error:  # noqa: PERF203 - retry
                last_error = error
                logger.warning(
                    "Falha na requisição (tentativa %d/%d): %s", attempt + 1, attempts, error
                )
                if isinstance(error, httpx.HTTPStatusError) and error.response.status_code < 500:
                    logger.warning(
                        "Corpo da resposta (HTTP %d): %s",
                        error.response.status_code,
                        error.response.text[:500],
                    )
                if attempt < attempts - 1:
                    self._sleep(self._config.backoff_s * 2**attempt)
        raise IngestionError(f"Falha ao acessar {url} após {attempts} tentativas") from last_error
